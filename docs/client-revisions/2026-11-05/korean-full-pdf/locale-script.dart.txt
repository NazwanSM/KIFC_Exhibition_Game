   testWidgets('review F03 $size ${locale.name}',(t)async{
    await launchScreen(t,size:size,locale:locale,page:'finalChallenge');
    await t.tap(find.byKey(const ValueKey('impact-cleanAir')));await t.pump();

      'source':'Flutter widget render; static mascot fallback', 'language':locale.name,'width':size.width,'height':size.height});
    await t.pumpWidget(const SizedBox.shrink());
   });
  }
 }
