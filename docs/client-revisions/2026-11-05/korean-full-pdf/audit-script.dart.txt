const output = 'output/client-review/2026-11-05/korean-full-pdf-validated';
Finder kind(String s) => find.byWidgetPredicate((w) => w.runtimeType.toString() == s);
final manifest = jsonDecode(File('design/quick-specs/korean-full-pdf-2026-11-05.json').readAsStringSync()) as Map<String,dynamic>;
final approved = <Map<String,dynamic>>[
  ...(manifest['items'] as List).cast<Map<String,dynamic>>(),
  ...(manifest['composed'] as List).cast<Map<String,dynamic>>().map((x)=> {...x,'text':(x['lines'] as List).join('\n')}),
];
final reports = <Map<String, Object?>>[];

Future<void> settle(WidgetTester t) async {
  await t.pump();
  await t.pump(const Duration(milliseconds: 650));
  await t.pump(const Duration(milliseconds: 100));
}

Future<void> audit(WidgetTester t, Size size, String state) async {
  await settle(t);
  final issues = <Map<String, Object?>>[];
  var paragraphs = 0;
  final approvedIds = <String>{};
  for (final element in find.byType(RichText).evaluate()) {
    final p = element.renderObject! as RenderParagraph;
    final text = p.text.toPlainText();
    if (!RegExp(r'[가-힣]').hasMatch(text)) continue;
    final rect = p.localToGlobal(Offset.zero) & p.size;
    if (!rect.overlaps(Offset.zero & size)) continue;
    var hidden = false;
    for (RenderObject? parent = p.parent; parent != null; parent = parent.parent) {
      if (parent is RenderOpacity && parent.opacity == 0) hidden = true;
      if (parent is RenderAnimatedOpacity && parent.opacity.value == 0) hidden = true;
    }
    if (hidden) continue;
    paragraphs++;
    final clean = text.replaceAll('\u2060','');
    for (final item in approved) {
      final expected = item['text'] as String;
      if (!clean.contains(expected)) continue;
      approvedIds.add(item['id'] as String);
      final joined = AppCopy.keepKoreanWordBreaks(expected);
      var start = text.indexOf(joined);
      if (start < 0) continue;
      for (final line in joined.split('\n')) {
        final boxes=p.getBoxesForSelection(TextSelection(baseOffset:start,extentOffset:start+line.length));
        if (boxes.map((b)=>(b.top*10).round()).toSet().length != 1) {
          issues.add({'word':'[PDF-LINE-WRAP]','text':clean,'id':item['id']});
        }
        start+=line.length+1;
      }
    }
    if (p.hasSize && text.isNotEmpty) {
      final allBoxes = p.getBoxesForSelection(TextSelection(baseOffset:0, extentOffset:text.length));
      final wordOutside = allBoxes.any((b) => b.left < -1 || b.right > p.size.width + 1) &&
          RegExp(r'\S+').allMatches(text).any((m) => p.getBoxesForSelection(
            TextSelection(baseOffset:m.start,extentOffset:m.end),
          ).any((b) => b.left < -1 || b.right > p.size.width + 1));
      if (wordOutside) {
        issues.add({'word':'[CLIPPED]', 'pieces':<String>[], 'text':text.replaceAll('\u2060',''),
          'keys':<String>[], 'paragraphRect':[rect.left,rect.top,rect.width,rect.height]});
      }
    }
    for (final match in RegExp(r'\S+').allMatches(text)) {
      final word = match[0]!;
      if (!RegExp(r'[가-힣]').hasMatch(word)) continue;
      final boxes = p.getBoxesForSelection(TextSelection(
        baseOffset: match.start, extentOffset: match.end,
      ));
      final lines = boxes.map((b) => (b.top * 10).round()).toSet();
      if (lines.length < 2) continue;
      final pieces = <String>[];
      int? lastLine;
      for (var cursor = match.start; cursor < match.end; cursor++) {
        final glyphs = p.getBoxesForSelection(TextSelection(baseOffset:cursor,extentOffset:cursor+1));
        if (glyphs.isEmpty) continue;
        final line = (glyphs.first.top * 10).round();
        if (line != lastLine) { pieces.add(''); lastLine = line; }
        pieces[pieces.length-1] += text.substring(cursor,cursor+1).replaceAll('\u2060','');
      }
      final keys = <String>[];
      element.visitAncestorElements((e) {
        if (e.widget.key is ValueKey) keys.add((e.widget.key as ValueKey).value.toString());
        return true;
      });
      issues.add({'word':word.replaceAll('\u2060',''), 'pieces':pieces,
        'text':text.replaceAll('\u2060',''), 'keys':keys,
        'paragraphRect':[rect.left,rect.top,rect.width,rect.height]});
    }
  }
  final errors = <String>[];
  for (var error = t.takeException(); error != null; error = t.takeException()) {
    errors.add(error.toString());
  }
  final name = '$state-${size.width.toInt()}x${size.height.toInt()}';
  final report = <String,Object?>{'state':state, 'width':size.width,'height':size.height,
    'paragraphs':paragraphs, 'approvedIds':approvedIds.toList(), 'issues':issues, 'layoutErrors':errors,
    'screenshot':'$name.png',
    'source':'Flutter widget render; geometry-based word-break audit; static mascot fallback'};
  reports.add(report);
  await captureScreen(t, name, report.map((k,v)=>MapEntry(k,v ?? '')));
  await t.runAsync(() async {
    await File('$output/audit.json').writeAsString(const JsonEncoder.withIndent('  ').convert(reports));
  });
  expect(issues,isEmpty,reason:'$state at $size');
  expect(errors,isEmpty,reason:'$state at $size');
}

void main() {
  setUpAll(() async {
    await loadGameFonts();
    await rootBundle.loadString('assets/audio/narration/reading_timing.json');
    await (FontLoader('NotoSansKR')..addFont(rootBundle.load('assets/fonts/NotoSansKR-Variable.ttf'))).load();
  });
  for (final size in const [Size(360,640),Size(411,731),Size(720,1280),Size(1080,1920),Size(2160,3840)]) {
    testWidgets('audit Korean word wrapping $size', (t) async {
      Future<void> launch(String page, {int? card}) async {
        await t.pumpWidget(const SizedBox.shrink());
        await launchScreen(t, size:size, locale:AppLocale.korean, page:page, cardIndex:card);
      }
      for (final page in ['welcome','triangleExplanation','risk','finalChallenge','score','closing']) {
        await launch(page);
        await audit(t,size,page);
      }
      await launch('triangleBuild');
      await audit(t,size,'assembly-initial');
      for (final index in [1,3,5]) {
        await dropAssembly(t,index,0);
        await audit(t,size,'assembly-wrong-$index');
      }
      for (final pair in [(0,0),(2,1),(4,2)]) {
        await dropAssembly(t,pair.$1,pair.$2);
        if (pair.$2 < 2) await audit(t,size,'assembly-partial-${pair.$2+1}');
      }
      await audit(t,size,'assembly-complete');
      for (var i=0;i<3;i++) {
        t.platformDispatcher.accessibilityFeaturesTestValue = const FakeAccessibilityFeatures(disableAnimations:true);
        await launch('karhutlaCards',card:i);
        await waitForReading(t);
        await audit(t,size,'reading-${i+1}');
        t.platformDispatcher.clearAccessibilityFeaturesTestValue();
      }
      for (final choices in [[0],[1],[0,1],[0,2]]) {
        await launch('risk');
        for (final choice in choices) {
          await t.tap(find.descendant(of:kind('_RiskCard').at(choice),matching:find.byType(InkWell)));
          await settle(t);
        }
        await audit(t,size,'risk-${choices.join('-')}');
      }
      for (final correct in [true,false]) {
        await launch('causeQuiz');
        for (var i=0;i<4;i++) {
          await audit(t,size,'cause-${i+1}-initial');
          final answer = i == 2 ? 1 : 0;
          await t.tap(kind('_ChoiceButton').at(correct ? answer : 1-answer));
          await audit(t,size,'cause-${i+1}-${correct ? 'correct' : 'wrong'}');
          if (i < 3) {
            await t.tapAt(const Offset(4,4));
            await settle(t);
          }
        }
      }
      await launch('peatCharacteristics');
      await audit(t,size,'peat-cards-front');
      for (var i=0;i<3;i++) {
        await t.tap(find.byKey(ValueKey('peat-card-$i')));
        await audit(t,size,'peat-cards-${i+1}-open');
      }
      final labels = ['haze','ecosystemDamage','carbonEmissions','subsidence','cleanAir','carbonStorage'];
      final cases = <List<int>>[[],[0,1,2,3],[4,5],[4],[5],[1,2],[0],[0,1],[0,1,2]];
      for (var i=0;i<cases.length;i++) {
        await launch('finalChallenge');
        for (final token in cases[i]) {
          await t.tap(find.byKey(ValueKey('impact-${labels[token]}')));
          await t.pump();
        }
        await t.tap(find.byKey(const ValueKey('final-finish-early')));
        await audit(t,size,'final-feedback-F${(i+1).toString().padLeft(2,'0')}');
        expect(find.byKey(const ValueKey('final-review-message')),findsOneWidget);
      }
      // Inspect audio settings as another player-facing text surface.
      await launch('welcome');
      await t.tap(find.byKey(const ValueKey('audio-settings')));
      await audit(t,size,'audio-settings');
      final covered=reports.where((x)=>x['width']==size.width).expand((x)=>(x['approvedIds'] as List).cast<String>()).toSet();
      expect(covered, containsAll(approved.map((x)=>x['id'])), reason:'All 34 PDF blocks exercised at $size');
      await t.pumpWidget(const SizedBox.shrink());
    });
  }
}
